
const table = document.querySelector('#table-users-body');
const form = document.querySelector('#user-form');
const userName = document.querySelector('#user-name');
const userEmail = document.querySelector('#user-email');
const userAdmin = document.querySelector('#user-admin');

// armazena a linha que está sendo editada no momento
let linhaEmEdicao = null;

if (form) {
  form.addEventListener('submit', saveData);
}

function saveData(e) {
  e.preventDefault();
  let user = userName.value;
  let email = userEmail.value;
  let admin = userAdmin && userAdmin.checked ? 'Sim' : 'Não';

  // Se houver uma linha em edição, atualiza as células dela em vez de criar uma
  if (linhaEmEdicao) {
    const celulas = linhaEmEdicao.querySelectorAll('td');
    celulas[1].textContent = user; // Atualiza Nome
    celulas[2].textContent = email; // Atualiza Email
    celulas[3].textContent = admin; // Atualiza Admin

    linhaEmEdicao = null; // Limpa o estado de edição
    alert('Usuário atualizado com sucesso!');
  } else {
    // Modo de criação normal (seu código original)
    let tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${'foto'}</td>
      <td>${user}</td>
      <td>${email}</td>
      <td>${admin}</td>
      <td>
        <div>
          <button class="btn-excluir">Excluir</button>
          <button class="btn-editar">Editar</button>
        </div>
      </td>
    `;

    const btnExcluir = tr.querySelector('.btn-excluir');
    const btnEditar = tr.querySelector('.btn-editar');

    btnExcluir.addEventListener('click', () => deletar(tr));
    btnEditar.addEventListener('click', () => editar(tr));

    table.appendChild(tr);
  }

  form.reset();
}


function deletar(linha) {
  // Se a linha que está sendo deletada for a que está em edição, limpa o formulário
  if (linha === linhaEmEdicao) {
    linhaEmEdicao = null;
    form.reset();
  }
  linha.remove();
  alert('Usuário excluído!');
}



// 1. Crie uma função dedicada a extrair os dados da linha
function obterDadosDaLinha(linha) {
    const celulas = linha.querySelectorAll('td');
    return {
        nome: celulas[1].textContent,
        email: celulas[2].textContent,
        isAdmin: celulas[3].textContent === 'Sim',
        celulas: celulas // Retorna as células caso precise atualizar o texto delas
    };
}

// 2. Use a função dentro do saveData (Modo Edição)
if (linhaEmEdicao) {
    const dados = obterDadosDaLinha(linhaEmEdicao);
    dados.celulas[1].textContent = user;
    dados.celulas[2].textContent = email;
    dados.celulas[3].textContent = admin;
    
    linhaEmEdicao = null;
    alert('Usuário atualizado com sucesso!');
}

// 3. Use a mesma função dentro do editar
function editar(linha) {
    linhaEmEdicao = linha;
    
    // Reaproveitando a lógica de buscar os dados das células
    const dados = obterDadosDaLinha(linha);
    
    userName.value = dados.nome;
    userEmail.value = dados.email;
    userAdmin.checked = dados.isAdmin;
    
    userName.focus();
}
